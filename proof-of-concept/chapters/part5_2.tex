\subsubsection{Build Pipeline}

\begin{table}[H]
    \centering
    \renewcommand{\arraystretch}{1.3}
    \setlength{\tabcolsep}{6pt}
    \begin{tabular}{|>{\RaggedRight\arraybackslash}p{2.8cm}|>{\RaggedRight\arraybackslash}p{2.8cm}|>{\RaggedRight\arraybackslash}p{4.8cm}|>{\RaggedRight\arraybackslash}p{4.8cm}|}
        \hline
        \rowcolor[rgb]{0.8, 0.9, 1.0}
        \textbf{Module} & \textbf{Loại} & \textbf{Sở hữu} & \textbf{Không sở hữu} \\ \hline

        \texttt{tsc} & Trình biên dịch &
        Chuyển đổi TS $\rightarrow$ JS, kiểm tra lỗi kiểu dữ liệu (type checking) &
        Đóng gói tệp cuối, quản lý tài nguyên tĩnh \\ \hline

        \texttt{lessc} & Trình tiền xử lý &
        Chuyển đổi Less $\rightarrow$ CSS, xử lý biến và hàm giao diện &
        Thực thi logic game, đóng gói tệp JS \\ \hline

        \texttt{build/} & Kho lưu trữ &
        Lưu trữ các tệp \texttt{.js} và \texttt{.css} đã biên dịch &
        Mã nguồn gốc (\texttt{src}), tệp đóng gói hoàn chỉnh \\ \hline

        \texttt{sw.js} & Trình trung gian &
        Chiến lược lưu đệm (caching), hỗ trợ chơi offline &
        Render giao diện, logic phía server \\ \hline

    \end{tabular}
    \caption{Build Pipeline}
\end{table}

\noindent \textbf{Ghi chú}: Dự án không sử dụng bundler. Đầu ra của \texttt{tsc} là các tệp \texttt{.js} riêng lẻ trong \texttt{build/}, không gom thành \texttt{client.js} duy nhất.
